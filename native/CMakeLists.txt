cmake_minimum_required(VERSION 3.22)
project(replaygain_scanner C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
file(GLOB SCANNER_SOURCES src/*.c src/*.cpp)
add_library(replaygain_scanner SHARED ${SCANNER_SOURCES})
target_include_directories(replaygain_scanner PRIVATE ${CMAKE_PREFIX_PATH}/include ${FFMPEG_DIR}/include)
target_link_directories(replaygain_scanner PRIVATE ${CMAKE_PREFIX_PATH}/lib ${FFMPEG_DIR}/lib)
target_link_libraries(replaygain_scanner ebur128 avformat avcodec avutil swresample tag z m log)
target_link_options(replaygain_scanner PRIVATE "-Wl,-z,max-page-size=16384")
